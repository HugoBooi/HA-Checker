async function searchPokemon() { let rawInput = document.getElementById('searchInput').value.toLowerCase().trim(); const container = document.getElementById('resultContainer'); if (!rawInput) { container.innerHTML = ''; return; } container.innerHTML = '

Searching Pokémon database...

'; const originalSearchInput = rawInput; // 1. Direct alias mapping for regional shorthand names const aliasMap = { 'darmanitan-galar': 'darmanitan-galar-standard', 'darmanitan-galaria': 'darmanitan-galar-standard', 'darmanitan-galarian': 'darmanitan-galar-standard', 'tauros-paldea': 'tauros-paldea-combat-breed', 'meowth-galar': 'meowth-galar', 'corsola-galar': 'corsola-galar' }; if (aliasMap[rawInput]) { rawInput = aliasMap[rawInput]; } try { // Try direct endpoint first (e.g. darmanitan-galar-standard or vulpix-alola) let res = await fetch(`https://pokeapi.co/api/v2/pokemon/${rawInput}`); let data; // 2. If direct fetch fails, strip regional/form tags to get base species name for fallback if (!res.ok) { const baseSpeciesQuery = rawInput.split('-')[0]; const speciesRes = await fetch(`https://pokeapi.co/api/v2/pokemon-species/${baseSpeciesQuery}`); if (!speciesRes.ok) throw new Error('Not found'); const speciesData = await speciesRes.json(); const defaultVariety = speciesData.varieties.find(v => v.is_default) || speciesData.varieties[0]; res = await fetch(defaultVariety.pokemon.url); data = await res.json(); } else { data = await res.json(); } const formName = data.name.toLowerCase(); const speciesName = data.species.name.toLowerCase(); const types = data.types.map(t => t.type.name); const haObj = data.abilities.find(a => a.is_hidden); const hasHiddenAbility = Boolean(haObj); const formattedName = formName.replace(/-/g, ' ').replace(/\b\w/g, l => l.toUpperCase()); // Check exact form name, raw input, base species name, or regional form entries in sheet const isOwned = ownedPokemonSet.has(formName) || ownedPokemonSet.has(originalSearchInput) || ownedPokemonSet.has(speciesName) || Array.from(ownedPokemonSet).some(item => item.startsWith(`${speciesName}-`)); // Format sprite filenames for Showdown assets let spriteName = formName; if (formName === 'nidoran-f' || formName === 'nidoran-m') { spriteName = formName.replace('-', ''); } else if (formName === 'darmanitan-galar-standard') { spriteName = 'darmanitan-galar'; } else if (formName === 'darmanitan-standard') { spriteName = 'darmanitan'; } const showdownSprite = `https://play.pokemonshowdown.com/sprites/gen5/${spriteName}.png`; const defaultSprite = data.sprites.front_default || 'https://raw.githubusercontent.com/PokeAPI/sprites/master/sprites/pokemon/0.png'; container.innerHTML = `

${formattedName}

${types.map(t => `${t}`).join('')}
${hasHiddenAbility ? ` Hidden Ability: ${haObj.ability.name.replace(/-/g, ' ')} ` : ` ${formattedName} does not have a Hidden Ability `}
${hasHiddenAbility ? ( isOwned ? `
✓ IN COLLECTION
` : `
✗ NOT IN COLLECTION
` ) : ''}
`; } catch (err) { container.innerHTML = `

No Pokémon found named "${originalSearchInput}". Check your spelling!

`; } }
